# Build faac static lib

git clone https://github.com/msys2/MINGW-packages
cd MINGW-packages/
cd mingw-w64-faac/
MINGW_ARCH=ucrt64 makepkg-mingw -sCL --nocheck
pacman -U mingw-w64-ucrt-x86_64-faac-1.50-1-any.pkg.tar.zst
